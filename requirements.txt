# ============================================================
# 思洋 · 个人管理台 —— 后端依赖
# ------------------------------------------------------------
# 上一版只用到 Flask 一个包（数据写死在 Python 里）。
# 这一版数据落到 SQLite，所以多了 Flask-SQLAlchemy 和它的底座 SQLAlchemy。
#
# 安装： pip install -r requirements.txt
# 如果下载慢，加国内镜像：
#   pip install -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple
# ============================================================

# ---- Web 框架本体 ----
Flask==3.1.3

# ---- 数据库 ----
# Flask-SQLAlchemy 提供 db.Model / db.session 这套写法；
# SQLAlchemy 是它的底座，app.py 里 import 了 sqlalchemy.orm.validates，所以也显式列出来。
Flask-SQLAlchemy==3.1.1
SQLAlchemy==2.0.54

# ---- Flask 自带的 WSGI 工具库 ----
# 密码哈希（generate_password_hash）就是从它来的。装 Flask 时会自动带上，
# 这里显式钉住版本，避免以后环境漂移导致哈希算法对不上。
Werkzeug==3.1.8

# ---- 生产环境要用的 WSGI 服务器 ----
# 本地开发用 python app.py 就够；部署到 Linux 服务器用它常驻。
# 后面那个分号是「平台标记」：gunicorn 不支持 Windows，
# 所以 Windows 上执行 pip install -r requirements.txt 会自动跳过它，不会报错。
gunicorn>=21.2.0; sys_platform != "win32"
